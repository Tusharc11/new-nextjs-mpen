# 1. Backup
sudo cp /etc/nginx/sites-enabled/mpencil-app /etc/nginx/sites-enabled/mpencil-app.backup

# 2. Add the buffer settings automatically
sudo sed -i '/proxy_pass http:\/\/localhost:3000;/a\        \n        proxy_buffer_size 16k;\n        proxy_buffers 4 16k;\n        proxy_busy_buffers_size 32k;' /etc/nginx/sites-enabled/mpencil-app

# 3. Test
sudo nginx -t

# 4. If test passes, reload
sudo systemctl reload nginx

# 5. Verify
sudo tail -20 /var/log/nginx/error.log
