# 1. Check your Nginx configuration
sudo cat /etc/nginx/sites-enabled/* | grep -E "server_name|proxy_pass|listen"

# 2. Test Nginx config
sudo nginx -t

# 3. Check for 502 errors in Nginx logs
sudo tail -50 /var/log/nginx/error.log | grep -E "502|upstream"

# 4. Check Nginx access logs for 502s
sudo grep "502" /var/log/nginx/access.log | tail -10
