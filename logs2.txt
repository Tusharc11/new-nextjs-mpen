# Run this all-in-one command
ssh ubuntu@your-ec2-ip << 'EOF'
echo "=== 1. PM2 Status ==="
pm2 status

echo -e "\n=== 2. Is Port 3000 Listening? ==="
sudo ss -tulpn | grep 3000

echo -e "\n=== 3. Test Local Connection ==="
curl -I http://localhost:3000/

echo -e "\n=== 4. Can We Reach External API? ==="
curl -I https://app.mpencil.io/ 2>&1 | head -5

echo -e "\n=== 5. Recent Application Logs ==="
pm2 logs nextjs-app --lines 30 --nostream 2>&1 | tail -30

echo -e "\n=== 6. Nginx Status ==="
sudo systemctl status nginx | head -3
EOF
